CREATE OR REPLACE VIEW V_S03MPER AS
SELECT A.cCodPer, A.cNombre, B.cTipo FROM S03MPER A INNER JOIN S03PSES B ON A.cCodPer = B.cCodPer;

CREATE OR REPLACE VIEW V_S03MFAC AS
(
SELECT A.*, B.cDescri AS cDesPrv, C.cDescri AS cDesMon, D.cDescri AS cDesSed, E.cNombre, F.cDescri AS cDesEst FROM S03MFAC A
INNER JOIN S03MPRV B ON A.cCodPrv = B.cCodPrv
LEFT OUTER JOIN V_S02TTAB C ON A.cMoneda = C.cCodigo AND C.cCodTab = '007'
LEFT OUTER JOIN V_S02TTAB D ON A.cSede = D.cCodigo AND D.cCodTab = '042'
INNER JOIN S03MPER E ON A.cCodPer = E.cCodPer
LEFT OUTER JOIN V_S02TTAB F ON A.cEstado = F.cCodigo AND F.cCodTab = '041'
);

CREATE OR REPLACE VIEW V_S03TUSU AS
(
   SELECT A.cCodUsu, A.cCodPer, A.cEstado, A.cClave, A.dFecFin, B.cNombre, B.cEmail, B.cArea
   FROM S03TUSU A INNER JOIN S03MPER B ON A.cCodPer = B.cCodPer
);
